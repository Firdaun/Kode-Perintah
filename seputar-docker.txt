// ===== Seputar Docker ===== //

membuat container(forward port)  : docker container create --name namabebas --publish 8080:80 redis:latest
membuat container                : docker container create --name contohredis2 redis:latest
hapus container                  : docker container rm contohredis
start container                  : docker container start contohredis
stop container                   : docker container stop contohredis
list semua container             : docker container ls -a
list container(hanya yg aktif)   : docker container ls
menambahkan env                  : docker container create --name contohmongo --publish 27017:27017 --env MONGO_INITDB_ROOT_USERNAME=fahrul --env MONGO_INITDB_ROOT_PASSWORD=fahrul mongo:latest
mengatur resource                : docker container create --name smallnginx --memory 100m --cpus 0.5 --publish 8081:80 nginx:lates
masuk ke container               : docker container exec -i -t contohredis /bin/bash
masuk ke redis lengkap           : redis-cli -h [IP_ADDRESS] -p 6379
masuk langsung ke redis-cli      : docker container exec -it redis-kustom redis-cli
masuk ke redis cli               : redis-cli --> get fahrul
bind mounts docker               : docker container create --name mongodata --publish 27017:27017 --mount "type=bind,source=C:\Users\Fahrul10\Desktop\mongodata,destination=/data/db" --env MONGO_INITDB_ROOT_USERNAME=fahrul --env MONGO_INITDB_ROOT_PASSWORD=fahrul mongo:latest
ngecek volume                    : docker volume ls
buat volume                      : docker volume create mongovolume
hapus volume                     : docker volume rm mongovolume
container volume                 : docker container create --name mongovolume --publish 27018:27017 --mount "type=volume,source=mongodata,destination=/data/db" --env MONGO_INITDB_ROOT_USERNAME=fahrul --env MONGO_INITDB_ROOT_PASSWORD=fahrul mongo:latest
pindahkan file ke windows        : docker container cp redis-kustom:/data/dump.rdb .
pindahkan file ke container      : docker container cp dump.rdb redis-kustom:/data/dump.rdb
membuat container(tanpa config)  : docker container create --publish 6379:6379 --mount "type=volume,source=brankas-redis,destination=/data" --name redis-kustom redis
menghapus container yg menyala   : docker container rm -f redis-kustom
membuat container(dengan config) : docker container create --name redis-kustom --publish 6379:6379 --mount "type=volume,source=brankas-redis,destination=/data" --mount "type=bind,source=C:\Users\Fahrul10\Desktop\redis-config,destination=/config" redis redis-server /config/redis.conf
mematikan semua container        : docker stop $(docker ps -q)
menyalakan docker                : docker compose up
mematikan semua container        : docker compose down
rebuild image                    : docker compose up --build

// ===== Docker : Menghapus Semua (Prune) ===== //
container                        : docker container prune
image                            : docker image prune
network                          : docker network prune
volume                           : docker volume prune
hapus container, network, image  : docker system prune
